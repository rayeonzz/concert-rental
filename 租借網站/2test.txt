    <script>
        const RENT_PER_DAY = 700;
        const LIGHTSTICK_PER_DAY = 200;

        function calculatePrice() {
            const start = new Date(document.getElementById('startDate').value);
            const end = new Date(document.getElementById('endDate').value);
            const ls1 = document.getElementById('lightstick').checked;
            const ls2 = document.getElementById('lightstick2').checked;

            if (start && end && end >= start) {
                const diffTime = Math.abs(end - start);
                const days = Math.ceil(diffTime / (1000 * 60 * 60 * 24)) + 1; // 含當天
                
                let dailyAddon = 0;
                if(ls1) dailyAddon += LIGHTSTICK_PER_DAY;
                if(ls2) dailyAddon += LIGHTSTICK_PER_DAY;

                const total = (RENT_PER_DAY + dailyAddon) * days;

                document.getElementById('rentalDays').innerText = `${days} 天`;
                document.getElementById('totalRent').innerText = `NT$ ${total.toLocaleString()}`;
                return { days, total };
            } else {
                document.getElementById('rentalDays').innerText = `0 天`;
                document.getElementById('totalRent').innerText = `NT$ 0`;
                return { days: 0, total: 0 };
            }
        }

        function handleFormSubmit(e) {
            e.preventDefault();
            const { days, total } = calculatePrice();
            if(days <= 0) {
                alert('請選擇正確的起訖日期！');
                return;
            }

            const booking = {
                id: 'RY' + Date.now().toString().slice(-6),
                name: document.getElementById('custName').value,
                lineId: document.getElementById('lineId').value,
                startDate: document.getElementById('startDate').value,
                endDate: document.getElementById('endDate').value,
                days: days,
                total: total,
                date: new Date().toLocaleString()
            };

            let orders = JSON.parse(localStorage.getItem('rayeonzz_orders') || '[]');
            orders.unshift(booking);
            localStorage.setItem('rayeonzz_orders', JSON.stringify(orders));

            alert(`預約已建立！單號：${booking.id}\n總金額：NT$ ${total}\n按下確定後將自動複製預約資料。`);
            
            const msg = `您好！我想預約 S24 Ultra 手機租借：\n預約單號：${booking.id}\n姓名：${booking.name}\nLINE ID：${booking.lineId}\n日期：${booking.startDate} ~ ${booking.endDate} (${days}天)\n預估租金：NT$ ${total}`;
            navigator.clipboard.writeText(msg);
            alert('預約明細已複製至剪貼簿！您可以直接貼給 rayeonzz 的 LINE 進行確認。');
        }

        function openAdminModal() {
            document.getElementById('adminModal').classList.remove('hidden');
        }

        function closeAdminModal() {
            document.getElementById('adminModal').classList.add('hidden');
        }

        function checkAdminPass() {
            const pass = document.getElementById('adminPassword').value;
            if(pass === 'admin123') {
                document.getElementById('adminAuth').classList.add('hidden');
                document.getElementById('adminContent').classList.remove('hidden');
                loadOrders();
            } else {
                alert('密碼錯誤！(預設密碼為 admin123)');
            }
        }

        function loadOrders() {
            const orders = JSON.parse(localStorage.getItem('rayeonzz_orders') || '[]');
            document.getElementById('orderCount').innerText = orders.length;
            const container = document.getElementById('orderList');
            
            if(orders.length === 0) {
                container.innerHTML = '<p class="text-slate-600 font-medium py-4 text-center">目前尚無預約單</p>';
                return;
            }

            container.innerHTML = orders.map(o => `
                <div class="p-3.5 bg-white/50 rounded-2xl border border-white/70 space-y-1 shadow-sm">
                    <div class="flex justify-between font-bold text-slate-900">
                        <span>#${o.id} - ${o.name} (LINE: ${o.lineId})</span>
                        <span class="text-purple-700">NT$ ${o.total}</span>
                    </div>
                    <div class="text-slate-700 text-[11px] font-medium">
                        📅 檔期：${o.startDate} ~ ${o.endDate} (${o.days} 天) | 提交時間：${o.date}
                    </div>
                </div>
            `).join('');
        }

        function clearOrders() {
            if(confirm('確定要清空所有測試訂單嗎？')) {
                localStorage.removeItem('rayeonzz_orders');
                loadOrders();
            }
        }
    </script>
